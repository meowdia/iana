# iana-perc

Type-safe, allocation-free Rust bindings for IANA `perc` registries.

[dependencies]

use iana_perc::perc::DatagramTransportLayerSecurityDtlsTunnelProtocolMessageTypesForPrivacyEnhancedConferencing;

assert_eq!(DatagramTransportLayerSecurityDtlsTunnelProtocolMessageTypesForPrivacyEnhancedConferencing::SUPPORTED_SRTP_PROTECTION_PROFILES.value(), 1);
